cmake_minimum_required(VERSION 3.20)
project(SouloTorrent LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${LIBTORRENT_SOURCE}" libtorrent)
add_library(SouloTorrentBridge STATIC "${BRIDGE_SOURCE}/SouloTorrent.mm")
target_include_directories(SouloTorrentBridge PUBLIC "${BRIDGE_SOURCE}/include")
target_link_libraries(SouloTorrentBridge PUBLIC torrent-rasterbar "-framework Foundation")
target_compile_options(SouloTorrentBridge PRIVATE -fobjc-arc)
