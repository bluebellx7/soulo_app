    func testLiveYinsuwInspection() async throws {
        let model = WebViewModel()
        let scene = try XCTUnwrap(UIApplication.shared.connectedScenes.first as? UIWindowScene)
        let previous = scene.windows.first(where: \.isKeyWindow)
        let window = UIWindow(windowScene: scene)
        window.rootViewController = UIHostingController(rootView: WebViewRepresentable(viewModel: model))
        window.makeKeyAndVisible()
        defer { window.isHidden = true; window.rootViewController = nil; model.releaseWebViewRuntime(); previous?.makeKeyAndVisible() }
        model.loadURL(URL(string: "https://www.yinsuw.cc/voddetail/dw977F/")!)
        try await Task.sleep(for: .seconds(20))
        let web = try XCTUnwrap(model.webView)
        let info = try await js("""
        JSON.stringify({url:location.href,title:document.title,ua:navigator.userAgent,platform:navigator.platform,groups:window.__souloTiledBanners.all().map(g=>({count:g.elements.length,bounds:g.bounds})),fixed:[...document.querySelectorAll('*')].filter(e=>getComputedStyle(e).position==='fixed').map(e=>({tag:e.localName,id:e.id,cls:e.className,html:e.outerHTML.slice(0,1200),rect:e.getBoundingClientRect().toJSON(),clip:getComputedStyle(e).clipPath})),scripts:[...document.scripts].map(e=>e.src),iframes:[...document.querySelectorAll('iframe')].map(e=>e.outerHTML)})
        """, web)
        print("YINSUW_REAL_INSPECTION", info)
        let sample = try await web.callAsyncJavaScript("""
          let frames=0,detectedFrames=0,visibleFrames=0,layers=0;const start=performance.now();
          return await new Promise(resolve=>{function sample(){
            frames++;const groups=window.__souloImageBanners.all(),els=groups.flatMap(g=>g.elements);layers=Math.max(layers,els.length);
            if(els.length===2)detectedFrames++;
            if(els.some(e=>{const r=e.getBoundingClientRect();return r.width>0&&r.height>0&&getComputedStyle(e).display!=='none'}))visibleFrames++;
            if(performance.now()-start<20000)requestAnimationFrame(sample);else resolve({frames,detectedFrames,visibleFrames,layers,url:location.href});
          }sample()});
        """, arguments: [:], in: nil, contentWorld: ManualAdBlockRuntime.world) as? [String: Any]
        print("YINSUW_REAL_STABILITY", sample ?? [:])
        XCTAssertGreaterThan(sample?["detectedFrames"] as? Int ?? 0, 100)
        XCTAssertEqual(sample?["layers"] as? Int, 2)
        XCTAssertEqual(sample?["visibleFrames"] as? Int, 0)
        let html = try await js("document.documentElement.outerHTML", web) as? String ?? ""
        let dom = XCTAttachment(string: html); dom.name="yinsuw-live-dom"; dom.lifetime = .keepAlways; add(dom)
        let shot = XCTAttachment(image: try await web.takeSnapshot(configuration:nil)); shot.name="yinsuw-live-auto"; shot.lifetime = .keepAlways; add(shot)
        let title = try await js("document.title", web) as? String ?? ""
        XCTAssertTrue(title.contains("八仙"))
    }
