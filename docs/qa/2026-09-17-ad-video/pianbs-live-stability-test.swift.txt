    func testLivePianbsMosaicStability() async throws {
        let model = WebViewModel()
        let scene = try XCTUnwrap(UIApplication.shared.connectedScenes.first as? UIWindowScene)
        let previous = scene.windows.first(where: \.isKeyWindow)
        let window = UIWindow(windowScene: scene)
        window.rootViewController = UIHostingController(rootView: WebViewRepresentable(viewModel: model))
        window.makeKeyAndVisible()
        defer { window.isHidden = true; window.rootViewController = nil; model.releaseWebViewRuntime(); previous?.makeKeyAndVisible() }
        model.loadURL(URL(string: "https://www.pianbs.com/html/226611.html")!)
        for _ in 0..<300 {
            if let web = model.webView, (try? await js("[...document.querySelectorAll('[style*=background-position]')].some(e=>getComputedStyle(e).clipPath !== 'none')", web)) as? Bool == true { break }
            try await Task.sleep(for: .milliseconds(100))
        }
        let web = try XCTUnwrap(model.webView)
        let sample = try await web.callAsyncJavaScript("""
            let frames=0, visibleFrames=0, detectedFrames=0, tags=new Set();
            const start=performance.now();
            return await new Promise(resolve=>{
              function sample(){
                frames++;
                const tiles=[...document.querySelectorAll('[style*=background-position]')].filter(e=>getComputedStyle(e).backgroundImage!=='none');
                if(tiles.length)detectedFrames++;
                tiles.forEach(e=>tags.add(e.localName));
                if(tiles.some(e=>{const s=getComputedStyle(e),r=e.getBoundingClientRect();return r.width>0&&r.height>0&&s.display!=='none'&&s.visibility!=='hidden'&&s.clipPath==='none';}))visibleFrames++;
                if(performance.now()-start<20000)requestAnimationFrame(sample);
                else resolve({frames,visibleFrames,detectedFrames,tags:[...tags],url:location.href});
              } sample();
            });
            """, arguments: [:], in: nil, contentWorld: ManualAdBlockRuntime.world) as? [String: Any]
        print("PIANBS_REAL_STABILITY", sample ?? [:])
        XCTAssertGreaterThan(sample?["detectedFrames"] as? Int ?? 0, 100)
        XCTAssertEqual(sample?["visibleFrames"] as? Int, 0)
        XCTAssertEqual(sample?["url"] as? String, "https://www.pianbs.com/html/226611.html")
    }
